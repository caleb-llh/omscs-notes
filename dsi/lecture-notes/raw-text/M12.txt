struct FieldVectorHasher {
 std::size_t operator()(const std::vector<Field> &fields) const {
  std::size_t hash = 0;
  for (const auto &field : fields) {
   std::hash<std::string> hasher;
   std::size_t fieldHash = 0;
   fieldHash = hasher(…);
   hash ^= fieldHash + 0x9e3779b9 + (hash << 6) + (hash >> 2);
  }
  return hash;
 }
};
