                    || ' - ' || Users.Username)
       AS UsersWhoLiked
FROM Interactions
JOIN Users ON Interactions.UserID = Users.UserID
WHERE Interactions.ReactionType = 'Like'
GROUP BY Interactions.PostID
ORDER BY TotalLikes DESC
LIMIT 1;
